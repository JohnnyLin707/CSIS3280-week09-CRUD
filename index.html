<?php

namespace CSIS3280;

class Application
{
    private DefaultController $defaultController;
    private ProductController $productController;
    // 注意：demo 没有声明 $userController，但下面 login 分支却用它

    public function __construct()
    {
        $this->defaultController = new DefaultController();
        $this->productController = new ProductController();
        // 老师没有在这里 new UserController()，属于待补的 bug
    }

    public function run(): void
    {
        $action = filter_input(INPUT_GET, 'action');
        $isPostSubmission = ($_SERVER['REQUEST_METHOD'] === 'POST');
        $username = $_SESSION['username'] ?? '';   // 为登录/权限判断预留，当前 demo 未使用

        switch ($action)
        {
            // ===== 新增：认证相关 =====
            case 'login':
                $formData = [
                    'username' => filter_input(INPUT_POST,'username'),
                    'password' => filter_input(INPUT_POST,'password'),
                ];
                if($isPostSubmission && $formData['username']){
                    $this->userController->processLogin($formData);   // 交给 UserController 校验
                }
                else{
                    $this->defaultController->homepage('wrong username or password');
                }
                break;

            case 'logout':
                unset($_SESSION['username']);       // 只清掉登录标记
                session_destroy();                  // 销毁整个会话
                $location ="/";
                header("Locatin: $location");       // 注意：原文拼成 Locatin，正确应为 Location
                break;

            // ===== 原来的空壳，现在补上 =====
            case 'create':
                $this->productController->create();   // 显示新建表单
                break;

            case 'processCreate':
                $description = filter_input(INPUT_POST,'description');
                $price = filter_input(INPUT_POST,'price',FILTER_SANITIZE_NUMBER_FLOAT, FILTER_FLAG_ALLOW_FRACTION);
                if($isPostSubmission && !empty($description) && !empty($price)){
                    $this->productController->processCreate($description,$price);
                }
                else{
                    $this->defaultController->error("Please enter your valid data");
                }
                break;

            case 'edit':
                $id = filter_input(INPUT_GET,'id',FILTER_SANITIZE_NUMBER_INT);
                if(!empty($id)){
                    $this->productController->edit($id);   // 第一阶段：加载记录到表单
                }
                else{
                    $this->defaultController->error("You need the id");
                }
                break;

            case 'processEdit':
                // 仍然是空的（课件后面才补）
                break;

            case 'products':
                $this->productController->list();   // starter 已有，保持不变
                break;

            case 'delete':
                $id = filter_input(INPUT_POST,'id',FILTER_SANITIZE_NUMBER_INT);
                if($isPostSubmission && !empty($id)){
                    $this->productController->delete($id);
                }
                else{
                    $this->defaultController->error("Please provide the id");
                }
                break;

            case 'deleteAll':
                break;   // 仍为空

            case 'show':
                break;   // 仍为空

            default:
                $this->defaultController->homepage('');   // 参数从无到有
        }

中文注释：为什么加 / 功能

  • create / edit / delete 从“空 break”变成“调用控制器”：starter 已经把方法签名和 case 留好，老师按 一个 action 一条线 逐个填，不一次性重构全项目。
  • filter_input() 而不是直接读 $_POST：课件明确要求；它顺便过滤输入，FILTER_SANITIZE_NUMBER_INT 保证 id 是整数，FILTER_FLAG_ALLOW_FRACTION 让价格保留小数。
  • $isPostSubmission：同一个 action 既负责“显示表单”也负责“处理提交”，用请求方法分流；破坏性操作（delete）必须是 POST。
  • processCreate 成功后才调控制器；否则走 error() 页，避免空数据入库。
  • $username = $_SESSION['username'] ?? '' 是为登录后做权限判断预留的；但 demo 里后面没用它，所以菜单隐藏了，后端 action 其实还没被真正保护。
  • homepage('') 是因为 DefaultController::homepage() 改成必须收一个错误参数；不传参数会报错。
  • demo 的待修点：① login 里 $this->userController 没有在构造函数里实例化；② Locatin 是 Location 的拼写错误；③ Product::validate() 已定义却没在这里调用；④ processEdit / deleteAll / show 仍为空。

    public static function validate(): Array
    {
        $errors = [];   // 收集所有错误，而不是一错就中断

        $description = filter_input(INPUT_POST, 'description');
        if(empty($description)){
            $errors[] = 'Please enter description';   // 描述必填

        FILTER_FLAG_ALLOW_FRACTION);

ProductRepository.php（新增部分）
    public function insert(string $description, float $price): int
    {
        if (NULL == $this->connection) return -1;   // 连接失败给错误码

        $sql  = 'INSERT INTO product (description, price) ';
        $sql .= 'VALUES (:desc, :price)';             // 命名占位符，防 SQL 注入
        $stmt = $this->connection->prepare($sql);
        $stmt->bindParam(':desc',$description);
        $stmt->bindParam(':price',$price);

        $status = $stmt->execute();
        if($status){
            return $this->connection->lastInsertId();   // 自增 id 交回上层做高亮
        }
        else{
            return -1;
        }
    }

    public function delete(int $id): bool
    {
        if (NULL == $this->connection) return -1;   // 注意：声明 bool 却返回 -1
        $sql = "DELETE from product WHERE id=:id";   // WHERE 限定只删一条
        $stmt = $this->connection->prepare($sql);
        $stmt->bindParam(':id',$id);
        $result = $stmt->execute();
        return $result;
    }

中文注释：为什么加 / 功能

  • prepare → bindParam → execute 是课件强调的标准三步；用户输入的描述和价格不拼进 SQL 字符串，避免 SQL 注入。
  • lastInsertId() 只在 INSERT 成功后有意义；把它返回给控制器，才有“刚新增的那行闪一下高亮”的效果。
  • 开头 if (NULL == $this->connection) return ... 是防御式编程：数据库连不上时，上层拿到 -1 或 false，不会出现致命错误。
  • DELETE ... WHERE id = :id 必须带 WHERE；否则就变成清空整张表（那是另一个 action deleteAll 的职责，且 demo 还没写）。
  • 小瑕疵：delete() 的返回类型写成 bool，但连接失败时返回 -1；更规范应返回 false。
