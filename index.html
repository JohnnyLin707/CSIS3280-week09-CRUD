Creating a new record
创建新记录
English original
  • Creating a new record is done using INSERT statement
  • Depending on the structure of your page, you may need to have a specific template to display the form to insert a new record
  • The Application class has two action cases to handle the Create functionality: 'create' and 'processCreate'. These two actions proceed to two controller methods
Application.php
class Application
{
    public function run(): void
    {
        // ...
        case 'create':
            $this->productController->create();
            break;

        case 'processCreate':
            $description = filter_input(INPUT_POST, 'description');
            $price = filter_input(INPUT_POST, 'price', FILTER_SANITIZE_NUMBER_FLOAT,
                FILTER_FLAG_ALLOW_FRACTION);

            if ($isPostSubmission && !empty($description) && !empty($price)) {
                $this->productController->processCreate($description, $price);
            } else {
                $this->defaultController->error(
                    'error - new product needs a description and price (via a POST request)');
            }
            break;
    }
}
中文翻译
  • 创建新记录使用 INSERT 语句完成
  • 根据页面的结构，你可能需要一个专门的模板来显示用于插入新记录的表单
  • Application 类有两个 action case 来处理创建功能：'create' 和 'processCreate'。这两个 action 会转到两个控制器方法
Application.php
class Application（前端控制器）
{
    public function run(): void
    {
        // ...
        // 'create' 动作：显示创建表单
        case 'create':
            $this->productController->create();
            break;

        // 'processCreate' 动作：处理表单提交
        case 'processCreate':
            // 使用 filter_input 安全地读取并过滤表单数据
            $description = filter_input(INPUT_POST, 'description');
            $price = filter_input(INPUT_POST, 'price', FILTER_SANITIZE_NUMBER_FLOAT,
                FILTER_FLAG_ALLOW_FRACTION);

            // 只有在是 POST 提交且数据不为空时才处理
            if ($isPostSubmission && !empty($description) && !empty($price)) {
                $this->productController->processCreate($description, $price);
            } else {
                // 否则转到错误页面
                $this->defaultController->error(
                    'error - new product needs a description and price (via a POST request)');
            }
            break;
    }
}

Creating a new record
创建新记录
English original
  • The submitted data needs to be processed using prepare → bindParam → execute. Notice that:
  • The insert() method was called from ProductController's processCreate() method
  • The processCreate() was called because the form was submitted
  • The form was displayed using ProductController's create() method
ProductRepository.php
class ProductRepository
{
    public function insert(string $description, float $price): int
    {
        if (NULL == $this->connection) return -1;

        $sql = 'INSERT INTO product (description, price)'
             . ' VALUES (:description, :price)';
        $stmt = $this->connection->prepare($sql);

        $stmt->bindParam(':description', $description);
        $stmt->bindParam(':price', $price);

        $success = $stmt->execute();

        if ($success) {
            return $this->connection->lastInsertId();
        } else {
            return -1;
        }
    }
}
中文翻译
  • 提交的数据需要使用 prepare → bindParam → execute 来处理。注意：
  • insert() 方法是由 ProductController 的 processCreate() 方法调用的
  • processCreate() 之所以被调用，是因为表单被提交了
  • 表单是由 ProductController 的 create() 方法显示的
ProductRepository.php
class ProductRepository
{
    public function insert(string $description, float $price): int
    {
        if (NULL == $this->connection) return -1;

        // 使用命名占位符，防止 SQL 注入
        $sql = 'INSERT INTO product (description, price)'
             . ' VALUES (:description, :price)';
        $stmt = $this->connection->prepare($sql);

        // 绑定参数
        $stmt->bindParam(':description', $description);
        $stmt->bindParam(':price', $price);

        $success = $stmt->execute();

        // 插入成功则返回最后插入的 id
        if ($success) {
            return $this->connection->lastInsertId();
        } else {
            return -1;
        }
    }
}

class ProductRepository
{
    public function findAll(): array
    {
        if (NULL == $this->connection) return [];
        $sql = 'SELECT * FROM product';
        $stmt = $this->connection->prepare($sql);
        $stmt->execute();
        $stmt->setFetchMode(\PDO::FETCH_CLASS, Product::class);
        $products = $stmt->fetchAll();
        return $products;
    }
}
中文翻译

class ProductRepository
{
    public function findAll(): array
    {
        if (NULL == $this->connection) return [];
        $sql = 'SELECT * FROM product';
        $stmt = $this->connection->prepare($sql);
        $stmt->execute();

ProductRepository.php
class ProductRepository
{
    public function deleteAll(): int
    {
        if (NULL == $this->connection) return 0;
        $sql = 'DELETE FROM product';
        $stmt = $this->connection->prepare($sql);
        $stmt->execute();
        $numRowsAffected = $stmt->rowCount();
        return $numRowsAffected;
    }

    public function delete(int $id): bool
    {
        if (NULL == $this->connection) return false;
        $sql = 'DELETE FROM product WHERE id = :id';
        $stmt = $this->connection->prepare($sql);
        $stmt->bindParam(':id', $id);
        $success = $stmt->execute();

    public function deleteAll(): int
    {

        $stmt->execute();
        $numRowsAffected = $stmt->rowCount();
        return $numRowsAffected;

        'password' => filter_input(INPUT_POST, 'password')
    ];
    if ($isPostSubmission && $formData['username']) {
        $this->userController->processLogin($formData);
    } else {
        // 登录失败，返回首页并显示错误消息
        $this->defaultController->homepage('Wrong username or password');
    }
    break;

case 'logout':
    // 清除用户名并销毁会话
    unset($_SESSION['username']);
    session_destroy();
    $location = '/';
    header("Location: $location");
    break;
